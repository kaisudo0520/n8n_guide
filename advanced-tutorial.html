n8n 進階教學：LLM 整合與 AI 自動化應用

本教學將探索 n8n 與人工智能技術的結合，特別是大型語言模型 (LLM) 的整合和應用。適合已經掌握 n8n 基礎和中級知識，希望進一步探索 AI 自動化可能性的用戶。通過本教學，您將學習如何利用 n8n 和 LLM 創建智能化的自動化工作流，實現更複雜、更強大的應用。

n8n 與 LLM 整合

大型語言模型 (LLM) 是當前 AI 領域最重要的技術之一，n8n 提供了多種方式與各種 LLM 服務整合。

LLM 輸出處理

處理 LLM 的輸出是整合過程中的重要部分：

解析 JSON 輸出

當要求 LLM 生成 JSON 格式的輸出時：

// Function 節點 - 解析 JSON 輸出
return items.map(item => {
  const response = item.json.choices[0].message.content;
  
  try {
    // 嘗試直接解析 JSON
    const parsedJson = JSON.parse(response);
    return { json: parsedJson };
  } catch (error) {
    // 如果直接解析失敗，嘗試提取 JSON 部分
    const jsonMatch = response.match(/```json\n([\s\S]*?)\n```/) || 
                      response.match(/\{[\s\S]*\}/);
    
    if (jsonMatch) {
      try {
        const extractedJson = JSON.parse(jsonMatch[0].replace(/```json\n|```/g, ''));
        return { json: extractedJson };
      } catch (e) {
        // 如果提取的 JSON 也無法解析，返回原始響應
        return { 
          json: { 
            error: 'Failed to parse JSON',
            rawResponse: response 
          } 
        };
      }
    } else {
      // 沒有找到 JSON 格式
      return { 
        json: { 
          error: 'No JSON found in response',
          rawResponse: response 
        } 
      };
    }
  }
});
提取結構化信息

從非結構化文本中提取信息：

// Function 節點 - 提取結構化信息
return items.map(item => {
  const response = item.json.choices[0].message.content;
  
  // 使用正則表達式提取信息
  const nameMatch = response.match(/名稱[：:]\s*(.*?)[\n\r]/);
  const emailMatch = response.match(/郵箱[：:]\s*(.*?)[\n\r]/);
  const phoneMatch = response.match(/電話[：:]\s*(.*?)[\n\r]/);
  
  return {
    json: {
      name: nameMatch ? nameMatch[1].trim() : null,
      email: emailMatch ? emailMatch[1].trim() : null,
      phone: phoneMatch ? phoneMatch[1].trim() : null,
      rawResponse: response
    }
  };
});

提示版本管理

在 n8n 中管理和優化提示模板：

使用工作流變量

將提示存儲為工作流變量：

// Function 節點 - 設置提示模板

$workflow.variables.systemPrompt = `你是一個專業的客服助手...`;

動態生成提示

根據上下文動態生成提示：

// Function 節點 - 動態生成提示
return items.map(item => {
  // 獲取基本模板
  const basePrompt = $workflow.variables.userPromptTemplate;
  
  // 根據用戶類型調整提示
  let customizedPrompt = basePrompt;
  if (item.json.userType === 'technical') {
    customizedPrompt += '\n請提供技術細節和專業解釋。';
  } else if (item.json.userType === 'beginner') {
    customizedPrompt += '\n請使用簡單的語言和類比解釋，避免技術術語。';
  }
  
  // 填充模板變量
  const filledPrompt = customizedPrompt
    .replace('{{question}}', item.json.question)
    .replace('{{productInfo}}', item.json.productInfo || '')
    .replace('{{knowledgeBase}}', item.json.knowledgeBase || '');
  
  return {
    json: {
      ...item.json,
      prompt: filledPrompt
    }
  };
});

文檔處理和嵌入生成

處理文檔並生成嵌入向量：

文檔分割

將長文檔分割為適合嵌入的片段：

// Function 節點 - 文檔分割
return items.map(item => {
  const text = item.json.text;
  const source = item.json.source || 'unknown';
  const category = item.json.category || 'general';
  
  // 定義分割參數
  const chunkSize = 1000; // 每個片段的最大字符數
  const overlap = 200;    // 重疊字符數
  
  // 分割文本
  const chunks = [];
  for (let i = 0; i < text.length; i += chunkSize - overlap) {
    const chunk = text.substring(i, i + chunkSize);
    
    // 只添加非空的有意義片段
    if (chunk.trim().length > 100) {
      chunks.push({
        content: chunk,
        source,
        category,
        position: i,
        metadata: {
          source,
          category,
          position: i,
          length: chunk.length
        }
      });
    }
  }
  
  return {
    json: {
      chunks,
      chunkCount: chunks.length,
      originalSource: source
    }
  };
});

相似度搜索和檢索

實現高效的相似度搜索：

向量相似度搜索

在向量數據庫中搜索相似內容：

// PostgreSQL 節點 - 向量搜索
// 操作：執行查詢
// 查詢：
SELECT 
  id, 
  content, 
  source, 
  category,
  1 - (embedding <=> $1) AS similarity
FROM 
  knowledge_base
WHERE 
  category = $2
ORDER BY 
  embedding <=> $1
LIMIT 5;
// 查詢參數：
// - $1: {{$json.embedding}}
// - $2: {{$json.category || 'general'}}
混合搜索策略

結合關鍵詞和向量搜索：

// PostgreSQL 節點 - 混合搜索
// 操作：執行查詢
// 查詢：
WITH keyword_matches AS (
  SELECT 
    id, 
    content, 
    source, 
    category,
    ts_rank(to_tsvector('english', content), to_tsquery('english', $1)) AS keyword_rank,
    1 - (embedding <=> $2) AS vector_similarity
  FROM 
    knowledge_base
  WHERE 
    to_tsvector('english', content) @@ to_tsquery('english', $1)
    OR category = $3
),
vector_matches AS (
  SELECT 
    id, 
    content, 
    source, 
    category,
    0 AS keyword_rank,
    1 - (embedding <=> $2) AS vector_similarity
  FROM 
    knowledge_base
  ORDER BY 
    embedding <=> $2
  LIMIT 10
)
SELECT 
  id, 
  content, 
  source, 
  category,
  keyword_rank,
  vector_similarity,
  (keyword_rank * 0.4 + vector_similarity * 0.6) AS combined_score
FROM 
  (SELECT * FROM keyword_matches UNION SELECT * FROM vector_matches) AS combined
ORDER BY 
  combined_score DESC
LIMIT 5;
// 查詢參數：
// - $1: {{$json.keywordQuery}}
// - $2: {{$json.embedding}}
// - $3: {{$json.category || 'general'}}

上下文增強和回答生成

使用檢索到的內容增強 LLM 回答：

上下文組裝

將檢索到的內容組裝為 LLM 上下文：

// Function 節點 - 組裝上下文
return items.map(item => {
  const query = item.json.query;
  const retrievedDocs = item.json.retrievedDocs || [];
  
  // 組裝上下文
  let context = '';
  if (retrievedDocs.length > 0) {
    context = '以下是與問題相關的信息：\n\n';
    
    retrievedDocs.forEach((doc, index) => {
      context += `[文檔 ${index + 1}] 來源: ${doc.source}\n`;
      context += `${doc.content}\n\n`;
    });
  } else {
    context = '沒有找到與問題直接相關的信息。';
  }
  
  return {
    json: {
      ...item.json,
      context,
      hasRelevantContext: retrievedDocs.length > 0
    }
  };
});

回答評估和改進

評估回答質量並改進：

// Function 節點 - 評估回答
return items.map(item => {
  const answer = item.json.choices[0].message.content;
  const query = item.json.query;
  const hasRelevantContext = item.json.hasRelevantContext;
  
  // 檢查不確定性標記
  const uncertaintyMarkers = [
    "我不確定", "可能", "也許", "我猜", "不太清楚",
    "需要更多信息", "無法確定", "難以判斷", "我不知道"
  ];
  
  let confidenceScore = 0.9; // 默認高信心
  
  // 降低包含不確定性標記的回答的信心分數
  for (const marker of uncertaintyMarkers) {
    if (answer.toLowerCase().includes(marker.toLowerCase())) {
      confidenceScore -= 0.2;
      break;
    }
  }
  
  // 如果沒有相關上下文，降低信心分數
  if (!hasRelevantContext) {
    confidenceScore -= 0.3;
  }
  
  // 檢查回答長度是否過短
  if (answer.length < 50) {
    confidenceScore -= 0.2;
  }
  
  // 限制最終分數範圍
  confidenceScore = Math.max(0, Math.min(1, confidenceScore));
  
  return {
    json: {
      ...item.json,
      answer,
      confidenceScore,
      needsImprovement: confidenceScore < 0.6
    }
  };
});

function isGreeting(message) {
  const greetings = ['你好', '嗨', '哈囉', '早安', '午安', '晚安', 'hi', 'hello'];
  return greetings.some(g => message.toLowerCase().includes(g.toLowerCase()));
}

function isProductQuery(message) {
  const productKeywords = ['產品', '商品', '購買', '價格', '多少錢', '功能', '規格'];
  return productKeywords.some(k => message.includes(k));
}

function isOrderStatus(message) {
  return message.includes('訂單') && 
         (message.includes('狀態') || message.includes('進度') || message.includes('何時'));
}

return items.map(item => {
  const query = item.json.message;
  const intent = item.json.intent;
  const conversationHistory = item.json.conversationHistory || [];
  const retrievedDocs = item.json.retrievedDocs || [];
  
  // 構建系統提示
  let systemPrompt = `你是一個專業、友善的客服助手。`;
  
  // 根據意圖調整提示
  if (intent === 'product_query') {
    systemPrompt += `\n請提供準確的產品信息，包括功能、價格和可用性。`;
  } else if (intent === 'support_request') {
    systemPrompt += `\n請提供清晰的故障排除步驟和解決方案。`;
  }
  
  systemPrompt += `\n請根據提供的知識庫信息回答用戶問題。如果知識庫中沒有相關信息，請誠實地表明你不知道，並提供聯繫人工客服的方式。`;
  
  // 構建知識庫上下文
  let knowledgeContext = '';
  if (retrievedDocs.length > 0) {
    knowledgeContext = '以下是與問題相關的信息：\n\n';
    
    retrievedDocs.forEach((doc, index) => {
      knowledgeContext += `[文檔 ${index + 1}] 來源: ${doc.source}\n`;
      knowledgeContext += `${doc.content}\n\n`;
    });
  } else {
    knowledgeContext = '沒有找到與問題直接相關的信息。';
  }
  
  // 構建完整提示
  const messages = [
    { role: 'system', content: systemPrompt },
    ...conversationHistory,
    { role: 'user', content: `問題：${query}\n\n知識庫信息：\n${knowledgeContext}` }
  ];
  
  return {
    json: {
      ...item.json,
      messages,
      hasKnowledgeContext: retrievedDocs.length > 0
    }
  };
});
回答質量評估

評估回答的質量和可信度：

// Function 節點 - 回答評估
return items.map(item => {
  const response = item.json.choices[0].message.content;
  const hasKnowledgeContext = item.json.hasKnowledgeContext;
  
  // 檢查不確定性標記
  const uncertaintyMarkers = [
    "我不確定", "可能", "也許", "我猜", "不太清楚",
    "需要更多信息", "無法確定", "難以判斷", "我不知道"
  ];
  
  let confidenceScore = 0.9; // 默認高信心
  
  // 降低包含不確定性標記的回答的信心分數
  for (const marker of uncertaintyMarkers) {
    if (response.toLowerCase().includes(marker.toLowerCase())) {
      confidenceScore -= 0.2;
      break;
    }
  }
  
  // 如果沒有相關知識庫上下文，降低信心分數
  if (!hasKnowledgeContext) {
    confidenceScore -= 0.3;
  }
  
  // 檢查回答長度是否過短
  if (response.length < 50) {
    confidenceScore -= 0.2;
  }
  
  // 限制最終分數範圍
  confidenceScore = Math.max(0, Math.min(1, confidenceScore));
  
  return {
    json: {
      ...item.json,
      response,
      confidenceScore,
      requiresHumanReview: confidenceScore < 0.6
    }
  };
});

      step: 'translate_to_system',

      targetLanguage: systemLanguage,

    });
  }
  
  // 添加 RAG 和 LLM 處理步驟

    step: 'process_query',

    text: needsTranslation ? '{{translatedText}}' : userMessage
  });
  
  if (needsTranslation) {
    processingSteps.push({
      step: 'translate_to_user',
      sourceLanguage: systemLanguage,
      targetLanguage: detectedLanguage,
      text: '{{processedResponse}}'
    });
  }
  
  return {
    json: {

      needsTranslation,
      processingSteps,
